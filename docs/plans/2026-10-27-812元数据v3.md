```python
def front_matter(text):
    if not text.startswith('---'):
        return {}
    out = {}
    for line in text.splitlines()[1:]:
        if line.strip() == '---':
            break
        if ':' in line:
            k, v = line.split(':', 1)
            out[k.strip()] = v.strip()
    return out
```

- [ ] **Step 2: 扩展 `parse_md` 返回关联表**

```python
def parse_md(text):
    chapters, points, assoc = [], [], {}
    mode = None
    for line in text.splitlines():
        if line.startswith('# '):
            if '章节表' in line:
                mode = 'chapter'
            elif '知识点表' in line:
                mode = 'point'
            elif '知识点关联表' in line:
                mode = 'meta'
            else:
                mode = None
            continue
        if not line.startswith('|'):
            continue
        cells = split_row(line)
        if cells and cells[0] in ('id',):
            continue
        if cells and all(set(c) <= set('-: ') for c in cells):
            continue
        if mode == 'chapter' and len(cells) == len(CHAP_COLS):
            c = dict(zip(CHAP_COLS, cells))
            c['tier'] = int(c['tier'])
            c['weight_center'] = float(c['weight_center'])
            c['entry_count'] = int(c['entry_count'])
            c['weight_range'] = _range(c['weight_range'])
            chapters.append(c)
        elif mode == 'point' and len(cells) == len(POINT_COLS):
            p = dict(zip(POINT_COLS, cells))
            p['keywords'] = _arr(p['keywords'])
            p['tags'] = _arr(p['tags'])
            p['source'] = _src(p['source'])
            points.append(p)
        elif mode == 'meta' and len(cells) == len(META_COLS):
            assoc[cells[0]] = {
                'prerequisites': [] if cells[1] == '—' else _arr(cells[1]),
                'related': [] if cells[2] == '—' else _arr(cells[2]),
                'confusables': _json_cell(cells[3]),
                'exam_evidence': _json_cell(cells[4]),
            }
    return chapters, points, assoc


def _json_cell(v):
    if not v or v == '—':
        return None
    return json.loads(v)
```

- [ ] **Step 3: `_src` 支持 `file=` / `section=` 覆盖**

```python
def _src(v):
    """'ref;basis;file=…;section=…;outline=…' → dict"""
    if not v:
        return {}
    parts = v.split(';')
    out = {'ref': parts[0].strip()}
    for p in parts[1:]:
        p = p.strip()
        if not p:
            continue
        if p.startswith('outline='):
            out['outline'] = p[len('outline='):]
        elif p.startswith('file='):
            out['file'] = p[len('file='):]
        elif p.startswith('section='):
            out['section'] = p[len('section='):]
        elif 'basis' not in out:
            out['basis'] = p
    return out
```

```python
def validate_v3(sid, d, errors, warns):
    pts = d['points']
    by_id = {p['id']: p for p in pts}
    seen_order = collections.defaultdict(set)
    all_q = collections.defaultdict(list)

    for p in pts:
        pid = p['id']
        o = p.get('order')
        if not isinstance(o, int) or o < 1:
            errors.append(f'E27 {sid}: {pid} order 非法 {o!r}')
        elif o in seen_order[p.get('chapter')]:
            errors.append(f'E27 {sid}: {p.get("chapter")} order 重复 {o}')
        else:
            seen_order[p.get('chapter')].add(o)

        for f in ('prerequisites', 'related'):
            for rid in p.get(f) or []:
                if rid == pid or rid not in by_id:
                    errors.append(f'E25 {sid}: {pid} {f} 指向无效 {rid}')
        for c in p.get('confusables') or []:
            rid = c.get('point_id')
            if rid and (rid == pid or rid not in by_id):
                errors.append(f'E25 {sid}: {pid} confusables 指向无效 {rid}')
            if not c.get('distinction'):
                errors.append(f'E25 {sid}: {pid} confusables 缺 distinction')

        ev = p.get('exam_evidence') or {}
        if ev.get('level') not in EVIDENCE_LEVELS:
            errors.append(f'E26 {sid}: {pid} exam_evidence.level 非法')
        if ev.get('level') != 'pending' and not ev.get('sources'):
            errors.append(f'E26 {sid}: {pid} exam_evidence 缺 sources')

        covered = set()
        for c in p.get('cards') or []:
            ct = c.get('card_type')
            if ct not in CARD_TYPES:
                errors.append(f'E21 {sid}: {pid} card_type 非法 {ct!r}')
            ap = c.get('answer_points')
            if not isinstance(ap, list) or not ap:
                errors.append(f'E22 {sid}: {pid} answer_points 为空')
            refs = c.get('unit_refs')
            n_units = len(p.get('units') or [])
            if not isinstance(refs, list) or not refs:
                errors.append(f'E23 {sid}: {pid} unit_refs 为空')
            else:
                for r in refs:
                    if not isinstance(r, int) or not (1 <= r <= n_units):
                        errors.append(f'E23 {sid}: {pid} unit_refs 越界 {r} (units={n_units})')
                    else:
                        covered.add(r)
                if len(refs) > 3:
                    warns.append(f'W18 {sid}: {pid} 单卡覆盖 {len(refs)} 个 unit')
            q = re.sub(r'\s+', '', c.get('question') or '')
            all_q[q].append(pid)
            if c.get('kind') in ('名解', '简答') and len(c.get('question') or '') > 120:
                warns.append(f'W15 {sid}: {pid} 设问过长 {len(c["question"])}')
            if c.get('kind') in ('简答', '论述') and len(ap or []) < 2:
                warns.append(f'W16 {sid}: {pid} answer_points 少于 2 条')
            a = re.sub(r'\s+', '', c.get('answer') or '')
            an = re.sub(r'\s+', '', c.get('analysis') or '')
            if a and an and a in an and len(a) / max(len(an), 1) > 0.8:
                warns.append(f'W19 {sid}: {pid} analysis 复述 answer')

        for i in range(1, len(p.get('units') or []) + 1):
            if i not in covered:
                warns.append(f'W17 {sid}: {pid} unit {i} 未被卡片覆盖')

    for q, owners in all_q.items():
        if q and len(owners) > 1:
            errors.append(f'E24 {sid}: 重复设问 {q[:30]}… 出现在 {owners}')
```

```python
#!/usr/bin/env python3
"""812 v3 质量审计：原子度、覆盖率、重复题、卡型分布。"""
import collections, json, pathlib, re, statistics, sys

ROOT = pathlib.Path(__file__).resolve().parent.parent.parent
META = ROOT / '知识点元数据'
OUT = META / 'research' / '812-audit.md'


def norm(s):
    return re.sub(r'\s+', '', s or '')


def audit(sid='812'):
    d = json.loads((META / 'subjects' / sid / f'{sid}.json').read_text(encoding='utf-8'))
    pts = d['points']
    cards = [c for p in pts for c in p.get('cards') or []]
    q = collections.Counter(norm(c.get('question')) for c in cards)
    dup = {k: v for k, v in q.items() if k and v > 1}
    cpp = [len(p.get('cards') or []) for p in pts]
    ctypes = collections.Counter(c.get('card_type') for c in cards)
    kinds = collections.Counter(c.get('kind') for c in cards)
    ev = collections.Counter((p.get('exam_evidence') or {}).get('level') for p in pts)
    covered, total = 0, 0
    for p in pts:
        n = len(p.get('units') or [])
        refs = {r for c in p.get('cards') or [] for r in (c.get('unit_refs') or [])}
        total += n
        covered += len([i for i in range(1, n + 1) if i in refs])
    lines = [
        f'# {sid} v3 质量审计', '',
        f'- 知识点：{len(pts)}', f'- 卡片：{len(cards)}',
        f'- 每点卡片 min/中位/max：{min(cpp)}/{statistics.median(cpp):.0f}/{max(cpp)}',
        f'- 重复设问：{len(dup)}',
        f'- unit 覆盖：{covered}/{total} ({covered / max(total, 1) * 100:.1f}%)', '',
        '## 卡片类型分布', '', '| card_type | 数量 |', '|---|---|',
    ]
    for k, v in ctypes.most_common():
        lines.append(f'| {k} | {v} |')
    lines += ['', '## 卷面题型分布', '', '| kind | 数量 |', '|---|---|']
    for k, v in kinds.most_common():
        lines.append(f'| {k} | {v} |')
    lines += ['', '## 证据等级分布', '', '| level | 点数 |', '|---|---|']
    for k, v in ev.most_common():
        lines.append(f'| {k} | {v} |')
    if dup:
        lines += ['', '## 重复设问清单', '', '| 设问 | 次数 |', '|---|---|']
        for k, v in sorted(dup.items(), key=lambda x: -x[1]):
            lines.append(f'| {k[:60]} | {v} |')
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text('\n'.join(lines) + '\n', encoding='utf-8')
    print('\n'.join(lines[:12]))
    return 1 if dup else 0


if __name__ == '__main__':
    sys.exit(audit())
```

```python
#!/usr/bin/env python3
"""812 v3 原子单元生成：4–15 条/点，30–70 字，一事一句。"""
import argparse, json, pathlib, re, sys, threading
from concurrent.futures import ThreadPoolExecutor, as_completed

sys.path.insert(0, str(pathlib.Path(__file__).resolve().parent))
import llm

ROOT = pathlib.Path(__file__).resolve().parent.parent.parent
META = ROOT / '知识点元数据'
PATH = META / 'subjects' / '812' / '812-units.jsonl'
LOCK = threading.Lock()

SYSTEM = (
    '你是植物生理学教学专家。把给定知识点的理解解析拆成原子记忆单元。'
    '每条 30–70 字、只讲一件事、句子完整、可脱离上下文独立记忆；'
    '保留关键术语/数字/公式；按「定义→结构/组成→机制/过程→条件→意义/应用→易错」排序；'
    '不照抄原句序，不编造原文没有的内容；化学式用 Unicode 角标。'
    '只输出 JSON：{"id":"...","units":["...","..."]}'
)


def build_prompt(p):
    return (
        f'知识点：{p["point"]}\n短标题：{p["title"]}\n'
        f'理解解析：{p["explain"]}\n考法：{p["exam"]}\n'
        f'要求：4–15 条；每条 30–70 字；一事一句。'
    )


def one(p):
    raw = llm.call_llm([
        {'role': 'system', 'content': SYSTEM},
        {'role': 'user', 'content': build_prompt(p)},
    ], max_tokens=3000)
    obj = llm.parse_reply(raw)
    if not obj or obj.get('id') != p['id']:
        return None
    units = [u.strip() for u in obj.get('units', []) if u.strip()]
    if not (3 <= len(units) <= 18):
        return None
    if any(not (20 <= len(u) <= 85) for u in units):
        return None
    return {'id': p['id'], 'units': units}


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--ids', nargs='*')
    ap.add_argument('--force', action='store_true', help='已有 unit 的点也重生成')
    ap.add_argument('--concurrency', type=int, default=4)
    args = ap.parse_args()
    d = json.loads((META / 'subjects/812/812.json').read_text(encoding='utf-8'))
    pts = [p for p in d['points'] if not args.ids or p['id'] in args.ids]
    done = {}
    if PATH.exists():
        for line in PATH.read_text(encoding='utf-8').splitlines():
            if line.strip():
                r = json.loads(line)
                done[r['id']] = r
    todo = [p for p in pts if args.force or p['id'] not in done]
    print(f'待生成 {len(todo)} / {len(pts)}')
    with ThreadPoolExecutor(max_workers=args.concurrency) as ex:
        futs = {ex.submit(one, p): p for p in todo}
        for f in as_completed(futs):
            p = futs[f]
            try:
                r = f.result()
            except Exception as e:
                print('ERR', p['id'], e)
                continue
            if not r:
                print('FAIL', p['id'])
                continue
            with LOCK:
                done[r['id']] = r
                with PATH.open('a', encoding='utf-8') as fh:
                    fh.write(json.dumps(r, ensure_ascii=False) + '\n')
                print('OK', p['id'], len(r['units']))
    print('完成', len(done))


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3
"""812 v3 原子卡生成：每点按原子度出卡，不设硬上限。"""
import argparse, json, pathlib, re, sys, threading
from concurrent.futures import ThreadPoolExecutor, as_completed

sys.path.insert(0, str(pathlib.Path(__file__).resolve().parent))
import llm

ROOT = pathlib.Path(__file__).resolve().parent.parent.parent
META = ROOT / '知识点元数据'
PATH = META / 'subjects' / '812' / '812-cards-v3.jsonl'
LOCK = threading.Lock()

KINDS = {'名解', '简答', '论述'}
CARD_TYPES = {'定义', '原理', '条件', '公式', '过程', '辨析', '应用', '易错', '记忆'}

SYSTEM = """你是云南农业大学 812《植物生理学》命题研究员，同时是 Anki 原子卡设计师。
给定知识点、理解解析、考法与原子单元，请输出 4 张以上原子化自测卡；数量由知识原子度决定，不设上限。
硬性要求：
1. 每张卡只考一个目标，能独立回答，答案可自检；禁止把多个独立知识点塞进一张卡。
2. kind ∈ {名解, 简答, 论述}：名解只用于术语定义；简答用于简述/为什么/有哪些/比较；论述只用于综合论证。
3. card_type ∈ {定义, 原理, 条件, 公式, 过程, 辨析, 应用, 易错, 记忆}。
4. answer_points 2–6 条，每条 ≤40 字；answer 先结论后要点；analysis 60–400 字，写依据/步骤/易错，不得复述答案。
5. unit_refs 是该卡覆盖的 units 下标（1 基），不得越界；一张卡最多覆盖 3 个 unit。
6. 每点至少 1 张名解、1 张简答；不要重复设问；辨析卡必须写「最关键区别」；易错卡必须写「错误说法→原因→正确判断」。
7. 化学式用 Unicode 角标（C₆H₁₂O₆、CO₂），SN1/SN2/E1/E2 保持原样。
只输出 JSON：{"id":"...","cards":[...]}"""


def prompt(p):
    units = p.get('units') or []
    unit_lines = '\n'.join(f'{i + 1}. {u}' for i, u in enumerate(units))
    return (
        f'知识点 id：{p["id"]}\n考点：{p["point"]}\n短标题：{p["title"]}\n'
        f'理解解析：{p["explain"]}\n考法：{p["exam"]}\n'
        f'关键词：{"/".join(p.get("keywords") or [])}\n'
        f'原子单元（unit_refs 用这里的序号）：\n{unit_lines}\n'
        f'请输出 4 张以上卡，按原子度决定，不设上限。'
    )


def validate(p, obj):
    if not obj or obj.get('id') != p['id']:
        return 'id mismatch'
    cards = obj.get('cards')
    if not isinstance(cards, list) or len(cards) < 2:
        return 'cards < 2'
    n = len(p.get('units') or [])
    qs = set()
    kinds = set()
    for c in cards:
        if c.get('kind') not in KINDS:
            return f'bad kind {c.get("kind")}'
        if c.get('card_type') not in CARD_TYPES:
            return f'bad card_type {c.get("card_type")}'
        for f in ('question', 'answer', 'analysis'):
            if not (c.get(f) or '').strip():
                return f'missing {f}'
        ap = c.get('answer_points')
        if not isinstance(ap, list) or not (2 <= len(ap) <= 6):
            return 'answer_points invalid'
        refs = c.get('unit_refs')
        if not isinstance(refs, list) or not refs or len(refs) > 3:
            return 'unit_refs invalid'
        if any(not isinstance(r, int) or not (1 <= r <= n) for r in refs):
            return 'unit_refs out of range'
        q = re.sub(r'\s+', '', c['question'])
        if q in qs:
            return 'duplicate question in batch'
        qs.add(q)
        kinds.add(c['kind'])
        c['front'] = c['question']
        c['back'] = c['answer']
        c['point_id'] = p['id']
    if not {'名解', '简答'} <= kinds:
        return 'missing 名解/简答'
    obj['cards'] = cards
    return None


def one(p):
    for attempt in range(3):
        raw = llm.call_llm([
            {'role': 'system', 'content': SYSTEM},
            {'role': 'user', 'content': prompt(p)},
        ], max_tokens=8000)
        obj = llm.parse_reply(raw)
        err = validate(p, obj)
        if not err:
            return obj, None
        last = err
    return None, last


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--chapters', nargs='*')
    ap.add_argument('--ids', nargs='*')
    ap.add_argument('--force', action='store_true', help='已有卡片的点也重生成')
    ap.add_argument('--concurrency', type=int, default=3)
    args = ap.parse_args()
    d = json.loads((META / 'subjects/812/812.json').read_text(encoding='utf-8'))
    pts = d['points']
    if args.chapters:
        pts = [p for p in pts if p['chapter'] in args.chapters]
    if args.ids:
        pts = [p for p in pts if p['id'] in args.ids]
    done = {}
    if PATH.exists():
        for line in PATH.read_text(encoding='utf-8').splitlines():
            if line.strip():
                r = json.loads(line); done[r['id']] = r
    todo = [p for p in pts if args.force or p['id'] not in done]
    print(f'待生成 {len(todo)} / {len(pts)}')
    fails = []
    with ThreadPoolExecutor(max_workers=args.concurrency) as ex:
        futs = {ex.submit(one, p): p for p in todo}
        for f in as_completed(futs):
            p = futs[f]
            try:
                obj, err = f.result()
            except Exception as e:
                obj, err = None, str(e)
            if not obj:
                fails.append((p['id'], err)); print('FAIL', p['id'], err); continue
            with LOCK:
                with PATH.open('a', encoding='utf-8') as fh:
                    fh.write(json.dumps(obj, ensure_ascii=False) + '\n')
                print('OK', p['id'], len(obj['cards']))
    if fails:
        print('失败', len(fails), '个：', fails)
        sys.exit(1)


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3
"""把 812-cards-v3.jsonl 合并为正式 812-cards.jsonl。"""
import collections, json, pathlib, re, shutil, sys

ROOT = pathlib.Path(__file__).resolve().parent.parent.parent
META = ROOT / '知识点元数据'
SRC = META / 'subjects/812/812-cards-v3.jsonl'
DST = META / 'subjects/812/812-cards.jsonl'
BAK = ROOT / '.build/backup/2026-10-27-812-v2/812-cards.orig.jsonl'
PREF = ['简答', '名解', '论述']


def norm(s):
    return re.sub(r'\s+', '', s or '')


def main():
    rows = {}
    for line in SRC.read_text(encoding='utf-8').splitlines():
        if line.strip():
            r = json.loads(line); rows[r['id']] = r
    all_points = {p['id'] for p in json.loads(
        (META / 'subjects/812/812.json').read_text(encoding='utf-8'))['points']}
    missing = sorted(all_points - set(rows))
    if missing:
        raise SystemExit(f'缺少卡片的知识点 {len(missing)} 个：{missing[:10]}')
    extra = sorted(set(rows) - all_points)
    if extra:
        raise SystemExit(f'卡片指向不存在的知识点：{extra[:10]}')
    if DST.exists() and not BAK.exists():
        shutil.copy2(DST, BAK)
    all_q = collections.Counter()
    out = []
    for pid in sorted(rows):
        cards = rows[pid]['cards']
        for i, c in enumerate(cards, 1):
            c['card_id'] = f'c{pid}-{i}'
            c['point_id'] = pid
            c['front'] = c.get('question', '')
            c['back'] = c.get('answer', '')
            q = norm(c.get('question'))
            all_q[q] += 1
            if all_q[q] > 1:
                raise SystemExit(f'重复设问：{pid} {c.get("question")}')
        kinds = []
        for c in cards:
            if c['kind'] not in kinds:
                kinds.append(c['kind'])
        pick = next((c for k in PREF for c in cards if c['kind'] == k), cards[0])
        listen = {
            'card_id': pick['card_id'], 'kind': pick['kind'],
            'script': f"设问：{pick['question']} 答案：{pick['answer']} 解析：{pick['analysis']}",
            'audio': 'pending',
        }
        out.append({'id': pid, 'cards': cards, 'exam_types': kinds, 'listen': listen})
    DST.write_text(''.join(json.dumps(r, ensure_ascii=False) + '\n' for r in out),
                   encoding='utf-8')
    print('finalize', len(out), 'points', sum(len(r['cards']) for r in out), 'cards')


if __name__ == '__main__':
    main()
```
