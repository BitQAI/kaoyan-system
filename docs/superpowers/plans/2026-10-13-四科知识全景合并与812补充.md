```python
#!/usr/bin/env python3
"""扫描 4 份源 HTML：结构边界 / 类使用 / CSS 选择器 / 内部链接 / 冲突。只读。"""
import re, json, hashlib, pathlib

ROOT = pathlib.Path(__file__).resolve().parent.parent
SRC = {
    'qj':   ROOT / '统考三科大纲知识点全景.html',
    'v2':   ROOT / '2027考研_101政治_204英语二_315化学农_考点级大纲深度调研_V2.html',
    'w315': ROOT / '2027考研_315化学农_章节权重深度调研.html',
    'w812': ROOT / '2027考研_云南农业大学812植物生理学_章节权重深度调研.html',
}

def body_of(html: str) -> str:
    m = re.search(r'<body[^>]*>(.*)</body>', html, re.S)
    return m.group(1) if m else html

def strip_tag_text(frag: str) -> str:
    return re.sub(r'\s+', ' ', re.sub(r'<[^>]+>', '', frag)).strip()

def split_by(body: str, tag: str):
    """按 <h2>/<h3> 切块，返回 [(level, title, id_or_None, start, end)]"""
    pat = re.compile(rf'<{tag}\b[^>]*>(.*?)</{tag}>', re.S)
    marks = [(m.start(), m.group(0), strip_tag_text(m.group(1)), m.group(0)) for m in pat.finditer(body)]
    out = []
    for i, (pos, raw, title, full) in enumerate(marks):
        idm = re.search(r'id="([^"]+)"', full)
        end = marks[i + 1][0] if i + 1 < len(marks) else len(body)
        out.append({'title': title, 'id': idm.group(1) if idm else None, 'start': pos, 'end': end})
    return out

def scan(key: str):
    html = SRC[key].read_text(encoding='utf-8')
    body = body_of(html)
    h2s = split_by(body, 'h2')
    h3s = split_by(body, 'h3')
    classes = {}
    for cl in re.findall(r'class="([^"]+)"', body):
        for c in cl.split():
            classes[c] = classes.get(c, 0) + 1
    style = '\n'.join(re.findall(r'<style[^>]*>(.*?)</style>', html, re.S))
    selectors = sorted(set(re.findall(r'(?m)^\s*([.#][\w-]+)', style)))
    hrefs = re.findall(r'href="#([^"]+)"', body)
    ids = re.findall(r'id="([^"]+)"', body)
    return {'h2': h2s, 'h3': h3s, 'classes': classes, 'selectors': selectors,
            'internal_hrefs': sorted(set(hrefs)), 'ids': ids,
            'chars': len(html), 'sha256': hashlib.sha256(html.encode()).hexdigest()}

def main():
    inv = {k: scan(k) for k in SRC}
    build = ROOT / '.build'
    build.mkdir(exist_ok=True)
    (build / 'inventory.json').write_text(json.dumps(inv, ensure_ascii=False, indent=1), encoding='utf-8')
    (build / 'source_hashes.json').write_text(
        json.dumps({k: v['sha256'] for k, v in inv.items()}, ensure_ascii=False, indent=1), encoding='utf-8')
    # 冲突报告：同名字段在两套样式里都定义
    name_sets = {k: set(v['selectors']) for k, v in inv.items()}
    conflicts = sorted((name_sets['qj'] & (name_sets['v2'] | name_sets['w315'] | name_sets['w812'])))
    lines = ['# 源文件结构清单', '']
    for k, v in inv.items():
        lines.append(f"## {k}  ({v['chars']} chars)")
        for h in v['h2']:
            lines.append(f"- h2 [{h['start']}-{h['end']}] id={h['id']}  {h['title'][:60]}")
        lines.append(f"- h3 数: {len(v['h3'])}; 类数: {len(v['classes'])}; 内部链接: {v['internal_hrefs'][:8]}...")
    lines += ['', '## 全景 vs V2 系 同名类的选择器交集', '']
    lines += [f'- {c}' for c in conflicts]
    (build / 'inventory.md').write_text('\n'.join(lines), encoding='utf-8')
    print('OK', {k: (len(v['h2']), len(v['h3'])) for k, v in inv.items()})
    print('class conflicts:', conflicts)

if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3
import re, json, hashlib, pathlib, sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
BUILD = ROOT / '.build'
OUT = ROOT / '2027考研_四科知识全景与章节权重.html'

def main():
    t = OUT.read_text(encoding='utf-8')
    fails = []
    # 1) 源文件未变
    hashes = json.loads((BUILD/'source_hashes.json').read_text())
    for name, h in hashes.items():
        p = ROOT / {'qj':'统考三科大纲知识点全景.html',
                    'v2':'2027考研_101政治_204英语二_315化学农_考点级大纲深度调研_V2.html',
                    'w315':'2027考研_315化学农_章节权重深度调研.html',
                    'w812':'2027考研_云南农业大学812植物生理学_章节权重深度调研.html'}[name]
        if hashlib.sha256(p.read_text(encoding='utf-8').encode()).hexdigest() != h:
            fails.append(f'源文件被改动: {p.name}')
    # 2) 锚点完整：每个 href="#x" 的 x 必须在 id 集合中
    ids = set(re.findall(r'id="([^"]+)"', t))
    orphan = sorted({h for h in re.findall(r'href="#([^"]+)"', t) if h not in ids})
    if orphan: fails.append(f'孤儿锚点: {orphan}')
    # 3) 无外部资源
    ext = re.findall(r'(?:src|href)="https?://[^"]+"|@import[^;]*https?://', t)
    if ext: fails.append(f'外部资源引用: {len(ext)} 处')
    # 4) 类定义完整：正文用到的类都有样式定义（允许白名单例外）
    used = set()
    for cl in re.findall(r'class="([^"]+)"', t): used |= set(cl.split())
    defined = set(re.findall(r'\.([\w-]+)', t[t.find('<style'):t.find('</style>')]))
    missing = sorted(used - defined)
    if missing: fails.append(f'无样式定义的类: {missing}')
    # 5) md5 抽样：5 个源章节的清洗后文本出现在输出中（取每份源文一个长句片段）
    samples = json.loads((BUILD/'md5_samples.json').read_text())  # Task5 Step2 生成
    for name, snippet in samples.items():
        if snippet not in t: fails.append(f'内容丢失（抽样）: {name}')
    # 6) 导航块未混入：输出中不得出现源侧边栏类
    for bad in ['class="wrap"', 'id="toc"']:
        if bad in t: fails.append(f'疑似源导航混入: {bad}')
    print('\n'.join(fails) if fails else 'ALL PASS')
    sys.exit(1 if fails else 0)

if __name__ == '__main__':
    main()
```
